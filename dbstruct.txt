-- 1. SERVERS
--    One row per Discord server — stores all bot config
CREATE TABLE servers (
    server_id       TEXT        PRIMARY KEY,
    server_name     TEXT        NOT NULL,
    prefix          TEXT        NOT NULL DEFAULT '-',
    max_tokens      INTEGER     NOT NULL DEFAULT 1024,
    temperature     FLOAT       NOT NULL DEFAULT 0.0,
    chunk_size      INTEGER     NOT NULL DEFAULT 500,
    chunk_overlap   INTEGER     NOT NULL DEFAULT 100,
    faiss_k         INTEGER     NOT NULL DEFAULT 15,
    bm25_k          INTEGER     NOT NULL DEFAULT 10,
    system_prompt   TEXT        DEFAULT NULL,
    added_at        TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at      TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
 
 
-- 2. UPLOADS
--    Every file or URL uploaded to a server — who did it, what it was
CREATE TABLE uploads (
    id              UUID        PRIMARY KEY DEFAULT gen_random_uuid(),
    server_id       TEXT        NOT NULL REFERENCES servers(server_id) ON DELETE CASCADE,
    uploaded_by     TEXT        NOT NULL,   -- Discord user ID
    username        TEXT        NOT NULL,   -- Discord username
    type            TEXT        NOT NULL CHECK (type IN ('url', 'pdf', 'contacts')),
    name            TEXT        NOT NULL,   -- URL string or filename
    chunks          INTEGER     NOT NULL DEFAULT 0,
    status          TEXT        NOT NULL DEFAULT 'ok' CHECK (status IN ('ok', 'failed')),
    error           TEXT        DEFAULT NULL,
    uploaded_at     TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
 
 
-- 3. ANALYTICS
--    One row per day per server — all key metrics in one place
CREATE TABLE analytics (
    id              UUID        PRIMARY KEY DEFAULT gen_random_uuid(),
    server_id       TEXT        NOT NULL REFERENCES servers(server_id) ON DELETE CASCADE,
    day             DATE        NOT NULL,
    total_questions INTEGER     NOT NULL DEFAULT 0,
    answered        INTEGER     NOT NULL DEFAULT 0,
    failed          INTEGER     NOT NULL DEFAULT 0,
    no_kb           INTEGER     NOT NULL DEFAULT 0,   -- asked before any upload
    avg_latency_ms  FLOAT       DEFAULT NULL,
    unique_users    INTEGER     NOT NULL DEFAULT 0,
    total_uploads   INTEGER     NOT NULL DEFAULT 0,
    chunks_added    INTEGER     NOT NULL DEFAULT 0,
    top_topic       TEXT        DEFAULT NULL,          -- most asked keyword that day
    UNIQUE (server_id, day)
);
 
 
-- INDEXES
CREATE INDEX idx_uploads_server   ON uploads(server_id);
CREATE INDEX idx_uploads_user     ON uploads(uploaded_by);
CREATE INDEX idx_analytics_server ON analytics(server_id, day DESC);
 
 
-- AUTO-UPDATE updated_at on servers
CREATE OR REPLACE FUNCTION set_updated_at()
RETURNS TRIGGER AS $$
BEGIN NEW.updated_at = NOW(); RETURN NEW; END;
$$ LANGUAGE plpgsql;
 
DROP TRIGGER IF EXISTS trg_servers_updated ON servers;
CREATE TRIGGER trg_servers_updated
    BEFORE UPDATE ON servers
    FOR EACH ROW EXECUTE FUNCTION set_updated_at();

CREATE TABLE IF NOT EXISTS channels (
    id          SERIAL PRIMARY KEY,
    server_id   TEXT NOT NULL,
    channel_id  TEXT NOT NULL UNIQUE,
    added_at    TIMESTAMP DEFAULT NOW()
);
