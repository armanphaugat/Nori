CREATE TABLE public.admin_sessions (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  discord_id text NOT NULL,
  refresh_token_hash text NOT NULL UNIQUE,
  issued_at timestamp with time zone NOT NULL DEFAULT now(),
  expires_at timestamp with time zone NOT NULL DEFAULT (now() + '7 days'::interval),
  revoked boolean NOT NULL DEFAULT false,
  user_agent text,
  ip_address text,
  CONSTRAINT admin_sessions_pkey PRIMARY KEY (id),
  CONSTRAINT admin_sessions_discord_id_fkey FOREIGN KEY (discord_id) REFERENCES public.admin_users(discord_id)
);
CREATE TABLE public.admin_users (
  discord_id text NOT NULL,
  username text NOT NULL,
  avatar text,
  email text,
  discord_access_token text NOT NULL,
  discord_refresh_token text NOT NULL,
  discord_token_expiry timestamp with time zone NOT NULL,
  created_at timestamp with time zone NOT NULL DEFAULT now(),
  last_login timestamp with time zone NOT NULL DEFAULT now(),
  CONSTRAINT admin_users_pkey PRIMARY KEY (discord_id)
);
CREATE TABLE public.analytics (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  server_id text NOT NULL,
  day date NOT NULL,
  total_questions integer NOT NULL DEFAULT 0,
  answered integer NOT NULL DEFAULT 0,
  failed integer NOT NULL DEFAULT 0,
  no_kb integer NOT NULL DEFAULT 0,
  avg_latency_ms double precision,
  unique_users integer NOT NULL DEFAULT 0,
  total_uploads integer NOT NULL DEFAULT 0,
  chunks_added integer NOT NULL DEFAULT 0,
  top_topic text,
  CONSTRAINT analytics_pkey PRIMARY KEY (id),
  CONSTRAINT analytics_server_id_fkey FOREIGN KEY (server_id) REFERENCES public.servers(server_id)
);
CREATE TABLE public.channels (
  id integer NOT NULL DEFAULT nextval('channels_id_seq'::regclass),
  server_id text NOT NULL,
  channel_id text NOT NULL UNIQUE,
  added_at timestamp without time zone DEFAULT now(),
  CONSTRAINT channels_pkey PRIMARY KEY (id),
  CONSTRAINT channels_server_id_fkey FOREIGN KEY (server_id) REFERENCES public.servers(server_id)
);
CREATE TABLE public.guild_admins (
  id integer NOT NULL DEFAULT nextval('guild_admins_id_seq'::regclass),
  guild_id text NOT NULL,
  discord_id text NOT NULL,
  role text NOT NULL CHECK (role = ANY (ARRAY['owner'::text, 'admin'::text])),
  granted_by text,
  granted_at timestamp with time zone NOT NULL DEFAULT now(),
  CONSTRAINT guild_admins_pkey PRIMARY KEY (id),
  CONSTRAINT guild_admins_guild_id_fkey FOREIGN KEY (guild_id) REFERENCES public.servers(server_id),
  CONSTRAINT guild_admins_discord_id_fkey FOREIGN KEY (discord_id) REFERENCES public.admin_users(discord_id),
  CONSTRAINT guild_admins_granted_by_fkey FOREIGN KEY (granted_by) REFERENCES public.admin_users(discord_id)
);
CREATE TABLE public.servers (
  server_id text NOT NULL,
  server_name text NOT NULL,
  prefix text NOT NULL DEFAULT '-'::text,
  max_tokens integer NOT NULL DEFAULT 1024,
  temperature double precision NOT NULL DEFAULT 0.0,
  chunk_size integer NOT NULL DEFAULT 500,
  chunk_overlap integer NOT NULL DEFAULT 100,
  faiss_k integer NOT NULL DEFAULT 15,
  bm25_k integer NOT NULL DEFAULT 10,
  system_prompt text,
  added_at timestamp with time zone NOT NULL DEFAULT now(),
  updated_at timestamp with time zone NOT NULL DEFAULT now(),
  mod_channel text,
  CONSTRAINT servers_pkey PRIMARY KEY (server_id)
);
CREATE TABLE public.uploads (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  server_id text NOT NULL,
  uploaded_by text NOT NULL,
  username text NOT NULL,
  type text NOT NULL CHECK (type = ANY (ARRAY['url'::text, 'pdf'::text, 'contacts'::text])),
  name text NOT NULL,
  chunks integer NOT NULL DEFAULT 0,
  status text NOT NULL DEFAULT 'ok'::text CHECK (status = ANY (ARRAY['ok'::text, 'failed'::text])),
  error text,
  uploaded_at timestamp with time zone NOT NULL DEFAULT now(),
  CONSTRAINT uploads_pkey PRIMARY KEY (id),
  CONSTRAINT uploads_server_id_fkey FOREIGN KEY (server_id) REFERENCES public.servers(server_id)
);
